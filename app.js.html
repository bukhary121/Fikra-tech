async function getLeads() {
  const city = document.getElementById("city").value;
  const type = document.getElementById("type").value;
  const limit = document.getElementById("limit").value;

  const query = `
  [out:json][timeout:25];
  area["name"="${city}"]->.searchArea;
  (
    node["amenity"="${type}"](area.searchArea);
    node["shop"](area.searchArea);
  );
  out ${limit};
  `;

  const res = await fetch("https://overpass-api.de/api/interpreter", {
    method: "POST",
    body: query
  });

  const json = await res.json();

  const table = document.getElementById("data");
  table.innerHTML = "";

  json.elements.forEach(el => {
    const name = el.tags?.name || "غير معروف";

    table.innerHTML += `
      <tr>
        <td>${name}</td>
        <td>
          <button onclick="sendToSheet('${name}')">حفظ</button>
          <button onclick="sendWhatsApp('${name}')">واتساب</button>
        </td>
      </tr>
    `;
  });
}

// 🔥 Google Sheets
async function sendToSheet(name) {
  const webhook = "PUT_YOUR_GOOGLE_SCRIPT_URL";

  await fetch(webhook, {
    method: "POST",
    body: JSON.stringify([{name: name}])
  });

  alert("تم الحفظ في Google Sheets");
}
